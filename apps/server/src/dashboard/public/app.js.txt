function node(tag, className, text) {
	const result = document.createElement(tag);
	if (className) result.className = className;
	if (text !== undefined) result.textContent = String(text);
	return result;
}

function button(label, callback, danger = false) {
	const result = node("button", `row-action${danger ? " danger" : ""}`, label);
	result.type = "button";
	result.addEventListener("click", callback);
	return result;
}

function showToast(message, error = false) {
	const toast = $("#toast");
	toast.textContent = message;
	toast.className = `toast visible${error ? " error" : ""}`;
	clearTimeout(toastTimer);
	toastTimer = setTimeout(() => {
		toast.className = "toast";
	}, 3500);
}

async function api(path, options = {}) {
	const method = (options.method || "GET").toUpperCase();
	const headers = { accept: "application/json", ...(options.headers || {}) };
	if (options.body !== undefined) headers["content-type"] = "application/json";
	if (
		!["GET", "HEAD", "OPTIONS"].includes(method) &&
		path !== "/login" &&
		state.csrfToken
	) {
		headers["x-csrf-token"] = state.csrfToken;
	}
	const response = await fetch(`${API}${path}`, {
		...options,
		method,
		headers,
		credentials: "same-origin",
		body:
			options.body === undefined || typeof options.body === "string"
				? options.body
				: JSON.stringify(options.body),
	});
	let payload = null;
	const text = await response.text();
	if (text) {
		try {
			payload = JSON.parse(text);
		} catch {
			payload = null;
		}
	}
	if (response.status === 401 && path !== "/login") {
		showLogin();
		throw new Error("Your operator session expired. Sign in again.");
	}
	if (!response.ok) {
		throw new Error(payload?.error || `Request failed (${response.status})`);
	}
	return payload;
}

}

function formatDate(value, includeTime = false) {
	if (!value) return "Never";
	const date = new Date(value);
	if (Number.isNaN(date.getTime())) return "Unknown";
	return new Intl.DateTimeFormat(undefined, {
		dateStyle: "medium",
		...(includeTime ? { timeStyle: "short" } : {}),
	}).format(date);
}

function dateTimeLocal(value) {
	if (!value) return "";
	const date = new Date(value);
	if (Number.isNaN(date.getTime())) return "";
	return new Date(date.getTime() - date.getTimezoneOffset() * 60_000)
		.toISOString()
		.slice(0, 16);
}

function initials(value) {
	return String(value || "K")
		.split(/\s+/)
		.slice(0, 2)
		.map((part) => part[0])
		.join("")
		.toUpperCase();
}

function customerFor(id) {
	return state.customers.find((customer) => customer.id === id);
}

function licenseStatus(license) {
	return license.status?.status || "expired";
}

function limit(value) {
	return value > 0 ? value.toLocaleString() : "Unlimited";
}

function summaryCard(label, value, copy) {
	const card = node("article", "summary-card");
	card.append(node("span", "", label), node("strong", "", value), node("small", "", copy));
	return card;
}

function metric(label, value) {
	const item = node("div", "metric");
	item.append(node("span", "", label), node("strong", "", value));
	return item;
}

function filteredLicenses() {
	const query = state.search.trim().toLowerCase();
	return state.licenses.filter((license) => {
		if (state.type && license.type !== state.type) return false;
		if (state.status && licenseStatus(license) !== state.status) return false;
		if (!query) return true;
		const customer = customerFor(license.customerId);
		return [license.keyPrefix, license.type, customer?.name, customer?.email]
			.filter(Boolean)
			.some((value) => String(value).toLowerCase().includes(query));
	});
}

function filteredCustomers() {
	const query = state.search.trim().toLowerCase();
	if (!query) return state.customers;
	return state.customers.filter((customer) =>
		[customer.name, customer.email, JSON.stringify(customer.metadata || {})]
			.filter(Boolean)
			.some((value) => String(value).toLowerCase().includes(query)),
	);
}

function recordActions(...actions) {
	const container = node("div", "row-actions");
	container.append(...actions.filter(Boolean));
	return container;
}

function renderLicenseCard(license) {
	const customer = customerFor(license.customerId);
	const card = node("article", "record-card");
	const header = node("header", "record-header");
	const identity = node("div", "record-identity");
	identity.append(node("span", "avatar", "K"));
	const title = node("div", "record-title");
	title.append(
		node("strong", "", customer?.name || "Unknown customer"),
		node("span", "", license.keyPrefix),
	);
	identity.append(title);
	const badges = node("div", "badges");
	const status = licenseStatus(license);
	badges.append(node("span", "badge", license.type), node("span", `badge ${status}`, status));
	header.append(identity, badges);

	const metrics = node("div", "record-metrics");
	metrics.append(
		metric("IP slots", limit(license.maxIps)),
		metric("Devices", limit(license.maxDevices)),
		metric("Sessions", limit(license.maxSessions)),
	);
	const metadata = node("div", "record-meta");
	if (license.type === "metered") {
		const activeMeters = (license.meters || []).filter((meter) => !meter.archivedAt);
		metadata.textContent = `${activeMeters.length} active meter${activeMeters.length === 1 ? "" : "s"}`;
	} else if (license.type === "subscription") {
		metadata.textContent = license.stripe
			? `Stripe ${license.stripe.status} · Paid through ${formatDate(license.stripe.paidThrough)}`
			: `Expires ${formatDate(license.expiresAt)}`;
	} else if (license.type === "trial") {
		metadata.textContent = license.trialStartedAt
			? `Trial started ${formatDate(license.trialStartedAt)}`
			: `${license.trialDurationMinutes || 0} minute trial · Not started`;
	} else {
		metadata.textContent = "No type-level expiry";
	}
	if (license.status?.reason) metadata.textContent += ` · ${license.status.reason}`;

	const footer = node("footer", "record-footer");
	footer.append(node("span", "", `Created ${formatDate(license.createdAt)}`));
	footer.append(
			recordActions(
			button("Access", () => openAccess(license)),
			button("Meters", () => openMeters(license)),
			state.capabilities.stripe && license.stripe
				? button("Sync Stripe", () => syncStripe(license))
				: null,
			state.capabilities.stripe && license.type === "subscription"
				? button(license.stripe ? "Unlink Stripe" : "Link Stripe", () => manageStripe(license))
				: null,
			button("Edit", () => openLicenseEditor(license)),
			status === "revoked"
				? button("Restore", () => restoreLicense(license))
				: button("Revoke", () => revokeLicense(license)),
			button("Rotate", () => rotateLicense(license)),
			button("End sessions", () => terminateSessions(license)),
			button("Delete", () => deleteLicense(license), true),
		),
	);
	card.append(header, metrics, metadata, footer);
	return card;
}

function renderCustomerCard(customer) {
	const licenses = state.licenses.filter((license) => license.customerId === customer.id);
	const card = node("article", "record-card");
	const header = node("header", "record-header");
	const identity = node("div", "record-identity");
	identity.append(node("span", "avatar", initials(customer.name)));
	const title = node("div", "record-title");
	title.append(node("strong", "", customer.name), node("span", "", customer.email || customer.id));
	identity.append(title);
	header.append(identity, node("span", "badge", `${licenses.length} licenses`));
	const metrics = node("div", "record-metrics");
	metrics.append(
		metric("Active", licenses.filter((item) => licenseStatus(item) === "active").length),
		metric("Expired", licenses.filter((item) => licenseStatus(item) === "expired").length),
		metric("Revoked", licenses.filter((item) => licenseStatus(item) === "revoked").length),
	);
	const metadata = node("div", "record-meta");
	const metadataKeys = Object.keys(customer.metadata || {});
	metadata.textContent = metadataKeys.length
		? `Metadata: ${metadataKeys.slice(0, 4).join(", ")}`
		: "No metadata";
	const footer = node("footer", "record-footer");
	footer.append(node("span", "", `Created ${formatDate(customer.createdAt)}`));
	footer.append(
		recordActions(
			button("Licenses", () => {
				state.view = "licenses";
				state.search = customer.name;
				$("#search").value = customer.name;
				render();
			}),
			button("Edit", () => openCustomerEditor(customer)),
			button("Delete", () => deleteCustomer(customer, licenses.length), true),
		),
	);
	card.append(header, metrics, metadata, footer);
	return card;
}

function renderRecords() {
	const isLicenses = state.view === "licenses";
	const records = isLicenses ? filteredLicenses() : filteredCustomers();
	const active = state.licenses.filter((item) => licenseStatus(item) === "active").length;
	const revoked = state.licenses.filter((item) => licenseStatus(item) === "revoked").length;
	$("#record-summary").replaceChildren(
		...(isLicenses
			? [
					summaryCard("Total licenses", state.licenses.length, "Issued by this instance"),
					summaryCard("Active", active, "Currently usable"),
					summaryCard("Revoked", revoked, "Operator or billing blocks"),
					summaryCard("Customers", state.customers.length, "Behind-the-scenes owners"),
				]
			: [
					summaryCard("Customers", state.customers.length, "Operator managed"),
					summaryCard("Licenses", state.licenses.length, "Across all customers"),
					summaryCard("Active licenses", active, "Currently usable"),
					summaryCard("With metadata", state.customers.filter((item) => Object.keys(item.metadata || {}).length).length, "Tagged records"),
				]),
	);
	$("#records-title").textContent = isLicenses ? "All licenses" : "All customers";
	$("#records-total").textContent = `${records.length} record${records.length === 1 ? "" : "s"}`;
	$("#records-grid").replaceChildren(
		...records.map((record) =>
			isLicenses ? renderLicenseCard(record) : renderCustomerCard(record),
		),
	);
	$("#empty-state").classList.toggle("hidden", records.length > 0);
	$("#type-filter").classList.toggle("hidden", !isLicenses);
	$("#status-filter").classList.toggle("hidden", !isLicenses);
}

function updateFilterOptions() {
	const customerSelect = $("#stats-customer");
	const customerValue = state.statsCustomerId;
	customerSelect.replaceChildren(new Option("All customers", ""));
	for (const customer of state.customers) {
		customerSelect.append(new Option(customer.name, customer.id));
	}
	customerSelect.value = customerValue;
	const licenseSelect = $("#stats-license");
	const licenseValue = state.statsLicenseId;
	licenseSelect.replaceChildren(new Option("All licenses", ""));
	for (const license of state.licenses) {
		if (customerValue && license.customerId !== customerValue) continue;
		const customer = customerFor(license.customerId);
		licenseSelect.append(
			new Option(`${license.keyPrefix} · ${customer?.name || "Unknown"}`, license.id),
		);
	}
	licenseSelect.value = licenseValue;
}

function filteredEvents() {
	const recent = [...(state.liveEvents || []), ...(state.statistics?.recent || [])];
	const ids = new Set();
	return recent
		.filter((event) => {
			if (!event?.id || ids.has(event.id)) return false;
			ids.add(event.id);
			if (state.statsCustomerId && event.customerId !== state.statsCustomerId) return false;
			if (state.statsLicenseId && event.licenseId !== state.statsLicenseId) return false;
			return true;
		})
		.slice(0, 100);
}

function eventTitle(event) {
	return String(event.type || "activity")
		.split(".")
		.map((part) => part.replaceAll("_", " "))
		.join(" · ");
}

function renderEventFeed() {
	const events = filteredEvents();
	const rows = events.map((event) => {
		const row = node("article", "event-row");
		row.append(node("span", `event-marker ${event.outcome || "info"}`));
		const copy = node("div", "event-copy");
		copy.append(node("strong", "", eventTitle(event)));
		const context = [
			event.keyPrefix,
			event.reason,
			event.source,
			formatDate(event.createdAt, true),
		]
			.filter(Boolean)
			.join(" · ");
		copy.append(node("span", "", context));
		row.append(copy);
		return row;
	});
	$("#event-feed").replaceChildren(
		...(rows.length ? rows : [node("div", "empty-state", "No activity in this filter")]),
	);
}

function renderStatistics() {
	updateFilterOptions();
	const statistics = state.statistics;
	if (!statistics) {
		$("#stats-totals").replaceChildren(summaryCard("Loading", "…", "Fetching statistics"));
		return;
	}
	const totalFor = (...types) =>
		(statistics.totals || [])
			.filter((total) => types.includes(total.type))
			.reduce((sum, total) => sum + Number(total.count || 0), 0);
	const activationAttempts = totalFor("license.activation_attempted");
	const activationSuccesses = totalFor("license.activation_succeeded");
	const heartbeats = totalFor("license.heartbeat");
	const usageConsumed = totalFor("usage.consumed");
	const usageRejected = totalFor("usage.rejected");
	$("#stats-totals").replaceChildren(
		summaryCard("Activation attempts", activationAttempts, `${activationSuccesses} succeeded`),
		summaryCard("Heartbeats", heartbeats, "Aggregated lifetime pings"),
		summaryCard("Usage events", usageConsumed, "Successful consumption calls"),
		summaryCard("Usage rejected", usageRejected, "Denied consumption calls"),
	);
	const groupedBuckets = new Map();
	for (const bucket of statistics.buckets || []) {
		const key = new Date(bucket.minute).toISOString();
		const grouped = groupedBuckets.get(key) || {
			minute: bucket.minute,
			activations: 0,
			heartbeats: 0,
			usage: 0,
		};
		if (bucket.type === "license.activation_attempted") grouped.activations += bucket.count;
		if (bucket.type === "license.heartbeat") grouped.heartbeats += bucket.count;
		if (bucket.type === "usage.consumed") grouped.usage += bucket.count;
		groupedBuckets.set(key, grouped);
	}
	const buckets = [...groupedBuckets.values()]
		.sort((left, right) => new Date(left.minute) - new Date(right.minute))
		.slice(-MAX_CHART_COLUMNS);
	const maximum = Math.max(
		1,
		...buckets.flatMap((bucket) => [bucket.activations, bucket.heartbeats, bucket.usage]),
	);
	const chart = buckets.map((bucket, index) => {
		const column = node("div", "chart-column");
		for (const [name, value] of [
			["activation", bucket.activations],
			["heartbeat", bucket.heartbeats],
			["usage", bucket.usage],
		]) {
			const bar = node("span", `chart-bar ${name}`);
			bar.style.height = `${Math.max(1, (Number(value) / maximum) * 100)}%`;
			bar.title = `${name}: ${value}`;
			column.append(bar);
		}
		if (index % Math.max(1, Math.ceil(buckets.length / 8)) === 0) {
			column.append(node("time", "", formatDate(bucket.minute)));
		}
		return column;
	});
	$("#activity-chart").replaceChildren(
		...(chart.length ? chart : [node("div", "empty-state", "No activity yet")]),
	);
	renderEventFeed();
}

function render() {
	const statistics = state.view === "statistics";
	const customers = state.view === "customers";
	$("#licenses-count").textContent = state.licenses.length;
	$("#customers-count").textContent = state.customers.length;
	for (const item of $$(".nav-item")) {
		item.classList.toggle("active", item.dataset.view === state.view);
	}
	$("#records-view").classList.toggle("hidden", statistics);
	$("#statistics-view").classList.toggle("hidden", !statistics);
	$("#add-button").classList.toggle("hidden", statistics);
	$("#view-title").textContent = statistics ? "Statistics" : customers ? "Customers" : "Licenses";
	$("#view-description").textContent = statistics
		? "Durable totals and a live stream of licensing activity."
		: customers
			? "Organize the owners behind product licenses."
			: "Manage product access without exposing keys to customers.";
	$("#add-button").textContent = customers ? "+ Add customer" : "+ Add license";
	if (statistics) renderStatistics();
	else renderRecords();
}

function metadataFrom(textarea) {
	const raw = textarea.value.trim();
	if (!raw) return {};
	const value = JSON.parse(raw);
	if (!value || typeof value !== "object" || Array.isArray(value)) {
		throw new Error("Metadata must be a JSON object.");
	}
	return value;
}

		);
	};
	$("#license-type").addEventListener("change", sync);
	sync();
	editorDialog.showModal();

function deleteCustomer(customer, licenseCount) {
	openConfirm(
		`Delete ${customer.name}?`,
		`This permanently deletes the customer and ${licenseCount} associated license${licenseCount === 1 ? "" : "s"}.`,
		"Delete customer",
		async () => {
			await api(`/customers/${encodeURIComponent(customer.id)}`, { method: "DELETE" });
			showToast("Customer deleted");
		},
	);
}

function deleteLicense(license) {
	openConfirm(
		"Delete this license?",
		`${license.keyPrefix} will stop working permanently. This cannot be undone.`,
		"Delete license",
		async () => {
			await api(`/licenses/${encodeURIComponent(license.id)}`, { method: "DELETE" });
			showToast("License deleted");
		},
	);

function accessPanel(title, listTitle, values, addLabel, addCallback, removeCallback) {
	const panel = node("section", "access-panel");
	panel.append(node("h3", "", title));
	if (addCallback) {
		const form = node("form", "inline-form");
		const input = node("input");
		input.name = "value";
		input.placeholder = addLabel;
		input.required = true;
		const submit = node("button", "button primary", "Add");
		submit.type = "submit";
		form.append(input, submit);
		form.addEventListener("submit", async (event) => {
			event.preventDefault();
			submit.disabled = true;
			try {
				await addCallback(input.value.trim());
				input.value = "";
			} catch (error) {
				$("#access-error").textContent = error.message;
			} finally {
				submit.disabled = false;
			}
		});
		panel.append(form);
	}
	panel.append(node("h4", "", listTitle));
	const list = node("div", "identifier-list");
	list.append(
		...(values.length
			? values.map((value) => identifierRow(value, () => removeCallback(value)))
			: [node("span", "muted", "None")]),
	);
	panel.append(list);
	return panel;
}

function attemptedAccessPanel(title, attempts, allowCallback) {
	const panel = node("section", "access-panel");
	panel.append(node("h3", "", title));
	const list = node("div", "identifier-list");
	for (const attempt of attempts) {
		const row = node("div", "identifier-row");
		const detail = node("div", "identifier-detail");
		detail.append(
			node("code", "", attempt.value),
			node(
				"span",
				"",
				`${attempt.attemptCount} activation attempt${attempt.attemptCount === 1 ? "" : "s"} · Last ${formatDate(attempt.lastAttemptedAt, true)}`,
			),
		);
		detail.title = `First attempted ${formatDate(attempt.firstAttemptedAt, true)}`;
		row.append(detail, button("Allow", () => allowCallback(attempt.value)));
		list.append(row);
	}
	if (!list.children.length) list.append(node("span", "muted", "No recent attempts"));
	panel.append(list);
	return panel;
}

async function loadAccess() {
	const id = encodeURIComponent(state.accessLicenseId);
	const access = await api(`/licenses/${id}/access`);
	const update = async (method, path, body) => {
		await api(`/licenses/${id}/access/${path}`, { method, body });
		await loadAccess();
	};
	const ipAllowlist = access.ipAllowlist || [];
	const deviceAllowlist = access.deviceAllowlist || [];
	const panels = [
		accessPanel(
			"Registered IP slots",
			"Automatically registered",
			(access.registeredIps || []).map((entry) => entry.value),
			"",
			null,
			(value) => update("DELETE", `registered-ips/${encodeURIComponent(value)}`),
		),
		accessPanel(
			"IP allowlist",
			"Allowed addresses",
			ipAllowlist,
			"IPv4 or IPv6 address",
			async (value) => {
				if (!ipAllowlist.length && !window.confirm("This is the first IP allowlist entry. All other IP addresses will be rejected. Continue?")) return;
				await update("POST", "ip-allowlist", { value });
			},
			(value) => update("DELETE", `ip-allowlist/${encodeURIComponent(value)}`),
		),
		attemptedAccessPanel(
			"Recent activation IPs",
			access.attemptedIps || [],
			async (value) => {
				if (!ipAllowlist.length && !window.confirm("Adding this IP creates the first IP allowlist entry. All other IP addresses will be rejected. Continue?")) return;
				await update("POST", "ip-allowlist", { value });
			},
		),
		accessPanel(
			"Registered device slots",
			"Automatically registered",
			(access.registeredDevices || []).map((entry) => entry.deviceId),
			"",
			null,
			(value) => update("DELETE", `registered-devices/${encodeURIComponent(value)}`),
		),
		accessPanel(
			"Device allowlist",
			"Allowed device IDs",
			deviceAllowlist,
			"Device ID",
			async (deviceId) => {
				if (!deviceAllowlist.length && !window.confirm("This is the first device allowlist entry. All other devices will be rejected. Continue?")) return;
				await update("POST", "device-allowlist", { deviceId });
			},
			(deviceId) => update("DELETE", `device-allowlist/${encodeURIComponent(deviceId)}`),
		),
		attemptedAccessPanel(
			"Recent activation devices",
			access.attemptedDevices || [],
			async (deviceId) => {
				if (!deviceAllowlist.length && !window.confirm("Adding this device creates the first device allowlist entry. All other devices will be rejected. Continue?")) return;
				await update("POST", "device-allowlist", { deviceId });
			},
		),
	];
	const sessions = node("section", "access-panel");
	sessions.append(
		node("h3", "", "Session controls"),
		node(
			"span",
			"muted",
			"Bearer session tokens stay opaque. Use End sessions on the license card to terminate every active session.",
		),
	);
	panels.push(sessions);
	$("#access-grid").replaceChildren(...panels);
}

async function adjustMeter(meter) {
	const deltaText = window.prompt(`Adjustment for ${meter.name} (negative debits, positive credits):`, "1");
	if (deltaText === null) return;
	const delta = Number(deltaText);
	if (!Number.isSafeInteger(delta) || delta === 0) {
		showToast("Adjustment must be a non-zero integer.", true);
		return;
	}
	const reason = window.prompt("Operator reason for the ledger:");
	if (!reason?.trim()) return;
	try {
		const meters = await api(`/licenses/${encodeURIComponent(state.meterLicenseId)}/meters/${encodeURIComponent(meter.name)}/adjust`, { method: "POST", body: { delta, reason: reason.trim() } });
		renderMeters(meters);
		showToast("Meter adjusted");
	} catch (error) {
		$("#meters-error").textContent = error.message;
	}
}

async function refreshStatisticsSnapshot() {
	try {
		await loadStatistics();
	} catch {
		// A later SSE reconnect or manual refresh will retry the durable snapshot.
	}
}

async function refresh(quiet = false) {
	const refreshButton = $("#refresh-button");
	refreshButton.disabled = true;
	try {
		const [customers, licenses] = await Promise.all([api("/customers"), api("/licenses")]);
		state.customers = customers;
		state.licenses = licenses;
		if (state.view === "statistics") await loadStatistics();
		render();
		if (!quiet) showToast("Dashboard refreshed");
	} catch (error) {
		showToast(error.message, true);
	} finally {
		refreshButton.disabled = false;
	}
}

function scheduleRealtimeRefresh() {
	clearTimeout(state.refreshTimer);
	state.refreshTimer = setTimeout(async () => {
		if (state.view === "statistics") {
			try {
				await loadStatistics();
			} catch {
				// The EventSource retry loop and manual refresh expose transient failures.
			}
		}
	}, 350);
}

function connectEvents() {
	if (state.eventSource || !state.capabilities.realtime) return;
	const source = new EventSource(`${API}/events`);
	state.eventSource = source;
	source.addEventListener("ready", () => {
		$("#live-dot").classList.add("online");
		$("#stream-state").classList.add("online");
		$("#stream-state").textContent = "Live";
		void refreshStatisticsSnapshot();
	});
	source.addEventListener("activity", (message) => {
		try {
			const activity = JSON.parse(message.data);
			if (!activity?.id || !activity?.type) return;
			state.liveEvents.unshift(activity);
			state.liveEvents = state.liveEvents.slice(0, 100);
			renderEventFeed();
			scheduleRealtimeRefresh();
		} catch {
			// Ignore malformed events; the durable snapshot remains authoritative.
		}
	});
	source.onerror = () => {
		$("#live-dot").classList.remove("online");
		$("#stream-state").classList.remove("online");
		$("#stream-state").textContent = "Reconnecting";
	};
}

$("#editor-form").addEventListener("submit", async (event) => {
	event.preventDefault();
	const submit = $("#editor-submit");
	submit.disabled = true;
	$("#editor-error").textContent = "";
	try {
		const form = new FormData(event.currentTarget);
		const { kind, record } = state.editing;
		if (kind === "customer") {
			const payload = {
				name: form.get("name"),
				email: form.get("email"),
				metadata: metadataFrom($("#customer-metadata")),
			};
			await api(record ? `/customers/${encodeURIComponent(record.id)}` : "/customers", {
				method: record ? "PATCH" : "POST",
				body: payload,
			});
			showToast(record ? "Customer updated" : "Customer created");
		} else {
			const type = form.get("type");
			const payload = {
				customerId: form.get("customerId"),
				type,
				maxIps: Number(form.get("maxIps")),
				maxDevices: Number(form.get("maxDevices")),
				maxSessions: Number(form.get("maxSessions")),
				metadata: metadataFrom($("#license-metadata")),
			};
			if (
				type === "subscription" &&
				!(record?.stripe && record.type === "subscription")
			) {
				payload.expiresAt = new Date(form.get("expiresAt")).toISOString();
			}
			if (type === "trial") payload.trialDurationMinutes = Number(form.get("trialDurationMinutes"));
			if (
				type === "metered" &&
				!record?.meters?.some((meter) => !meter.archivedAt)
			) {
				payload.meters = [{ name: form.get("meterName"), balance: Number(form.get("meterBalance")), reason: form.get("meterReason") }];
			}
			if (record?.stripe && type !== "subscription") {
				if (!window.confirm("Changing this license away from subscription unlinks Stripe without modifying the Stripe subscription. Continue?")) return;
				payload.confirmStripeUnlink = true;
			}
			const result = await api(record ? `/licenses/${encodeURIComponent(record.id)}` : "/licenses", {
				method: record ? "PATCH" : "POST",
				body: payload,
			});
			if (!record && result.licenseKey) showSecret(result.licenseKey);
			showToast(record ? "License updated" : "License created");
		}
		editorDialog.close();
		await refresh(true);
	} catch (error) {
		$("#editor-error").textContent = error.message;
	} finally {
		submit.disabled = false;
	}
});
